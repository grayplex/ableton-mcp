<objective>
Add a short TTL cache (30 seconds) to `get_checkpoint()` so that high-frequency agent
loops calling `get_next_actions` without an explicit `phase_name` do not re-query Ableton
on every call. The checkpoint reads live state via N+2 socket round-trips; caching the
result for 30s means repeated calls within an agent loop get instant responses.

Purpose: Agent loops calling `get_next_actions` without `phase_name` currently pay the
full checkpoint cost (~10 round-trips) on every invocation. A 30-second TTL cache
eliminates this for rapid successive calls while still reflecting state changes within
a reasonable window.

Output: Updated checkpoint.py with TTL cache, invalidation function, and passing tests.
</objective>

**3. checkpoint.py — add invalidation function:**

```python
def invalidate_checkpoint_cache(genre: str = None):
    """Clear cached checkpoint for a genre, or all if genre is None."""
    if genre is None:
        _checkpoint_cache.clear()
    else:
        _checkpoint_cache.pop(genre, None)
```

class TestCheckpointCache:
    def setup_method(self):
        """Clear cache before each test."""
        from MCP_Server.orchestration.checkpoint import _checkpoint_cache
        _checkpoint_cache.clear()

    def test_second_call_uses_cache(self):
        """Second get_checkpoint call within TTL should not call send_command again."""
        tracks = [_make_track("Drums", True, 0, has_clips=True),
                  _make_track("Bass", True, 1)]
        mock = _make_conn(tracks)
        with patch("MCP_Server.orchestration.checkpoint.get_ableton_connection",
                   return_value=mock):
            result1 = get_checkpoint("house")
            result2 = get_checkpoint("house")
        # send_command should only be called for the first invocation
        assert mock.send_command.call_count == 2  # get_arrangement_state + get_mix_state (only once)
        assert result1 == result2

    def test_different_genre_not_cached(self):
        """Different genre keys should get independent cache entries."""
        tracks = [_make_track("Drums", True, 0, has_clips=True),
                  _make_track("Bass", True, 1)]
        mock = _make_conn(tracks)
        with patch("MCP_Server.orchestration.checkpoint.get_ableton_connection",
                   return_value=mock):
            get_checkpoint("house")
            get_checkpoint("techno")
        # Should make 4 send_command calls (2 per genre)
        assert mock.send_command.call_count == 4

    def test_invalidate_forces_refresh(self):
        """invalidate_checkpoint_cache should force a fresh query."""
        from MCP_Server.orchestration.checkpoint import invalidate_checkpoint_cache
        tracks = [_make_track("Drums", True, 0, has_clips=True),
                  _make_track("Bass", True, 1)]
        mock = _make_conn(tracks)
        with patch("MCP_Server.orchestration.checkpoint.get_ableton_connection",
                   return_value=mock):
            get_checkpoint("house")
            invalidate_checkpoint_cache("house")
            get_checkpoint("house")
        # Should make 4 send_command calls (2 + 2, cache was invalidated)
        assert mock.send_command.call_count == 4

    def test_error_not_cached(self):
        """Connection errors should not be cached."""
        from MCP_Server.orchestration.checkpoint import _checkpoint_cache
        with patch("MCP_Server.orchestration.checkpoint.get_ableton_connection",
                   side_effect=Exception("connection refused")):
            result = get_checkpoint("house")
        assert "error" in result
        assert "house" not in _checkpoint_cache
